from flask import Flask, request, jsonify import sqlite3 from users import get_user_name app = Flask(__name__) DB = "database.db" def query_db(query, args=(), one=False): conn = sqlite3.connect(DB) cur = conn.cursor() cur.execute(query, args) rv = cur.fetchall() conn.commit() conn.close() return (rv[0] if rv else None) if one else rv # Lấy danh sách bài viết @app.route("/posts", methods=["GET"]) def get_posts(): rows = query_db("SELECT id, user_id, title, content FROM posts ORDER BY id DESC") posts = [{"id": r[0], "author": get_user_name(r[1]), "title": r[2], "content": r[3]} for r in rows] return jsonify(posts) # Tạo bài viết mới @app.route("/posts", methods=["POST"]) def create_post(): data = request.json query_db("INSERT INTO posts (user_id, title, content) VALUES (?, ?, ?)", (data['user_id'], data['title'], data['content'])) return jsonify({"status": "ok"}) # Lấy bình luận theo bài viết @app.route("/posts//comments", methods=["GET"]) def get_comments(post_id): rows = query_db("SELECT user_id, content FROM comments WHERE post_id=? ORDER BY id ASC", (post_id,)) comments = [{"author": get_user_name(r[0]), "content": r[1]} for r in rows] return jsonify(comments) # Thêm bình luận @app.route("/posts//comments", methods=["POST"]) def add_comment(post_id): data = request.json query_db("INSERT INTO comments (post_id, user_id, content) VALUES (?, ?, ?)", (post_id, data['user_id'], data['content'])) return jsonify({"status": "ok"}) if __name__ == "__main__": app.run(debug=True)
